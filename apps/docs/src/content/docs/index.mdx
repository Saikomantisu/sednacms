import { Card, CardGrid, LinkCard } from '@astrojs/starlight/components';

<CardGrid>
  <Card title="Forms from your fields" icon="pencil">
    Define fields in TypeScript. Sedna builds the forms at `/admin`, where editors
    can write posts, upload images, and publish content.
  </Card>
  <Card title="Publish without a rebuild" icon="rocket">
    Content lives in Postgres. Astro live collections read published entries on
    demand, so the next request can show an edit without a deployment.
  </Card>
  <Card title="Inside your Astro site" icon="astro">
    Install an integration and a server adapter. The admin, API, and public pages
    run in the same application.
  </Card>
  <Card title="Your database and storage" icon="database">
    Store content in Postgres and images in UploadThing or S3-compatible storage.
    With CDN caching enabled, published changes purge pages that read the collection.
  </Card>
</CardGrid>
